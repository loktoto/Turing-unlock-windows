cmake_minimum_required(VERSION 3.16)
project(tu102_host C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(SANITIZE "ASan and UBSan" OFF)
option(FUZZ "Build Clang libFuzzer target" OFF)
if(SANITIZE AND MSVC)
 message(FATAL_ERROR "SANITIZE requires the Linux GCC/Clang configuration; use OFF for Windows.")
endif()
if(FUZZ AND (MSVC OR NOT CMAKE_C_COMPILER_ID STREQUAL "Clang"))
 message(FATAL_ERROR "FUZZ requires Clang with the Unix compiler driver and libFuzzer.")
endif()
if(SANITIZE AND NOT MSVC)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(tu102_host STATIC host/layout.c host/vbios.c host/prep.c host/boot.c)
target_include_directories(tu102_host PUBLIC host)
if(MSVC)
 target_compile_options(tu102_host PRIVATE /W4)
else()
 target_compile_options(tu102_host PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_executable(host_tests tests/test_host.c)
target_link_libraries(host_tests PRIVATE tu102_host)
add_test(NAME host_contracts COMMAND host_tests)
if(FUZZ)
 target_compile_options(tu102_host PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
 add_executable(fuzz_parsers tests/fuzz.c)
 target_link_libraries(fuzz_parsers PRIVATE tu102_host)
 target_compile_options(fuzz_parsers PRIVATE -fsanitize=fuzzer,address,undefined)
 target_link_options(fuzz_parsers PRIVATE -fsanitize=fuzzer,address,undefined)
 target_link_options(host_tests PRIVATE -fsanitize=address,undefined)
endif()
add_executable(fwsec_tests tests/test_fwsec.c)
target_link_libraries(fwsec_tests PRIVATE tu102_host)
if(FUZZ)
 target_link_options(fwsec_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME fwsec_contracts COMMAND fwsec_tests)
add_library(tu102_platform_model STATIC host/platform_model.c)
target_include_directories(tu102_platform_model PUBLIC host)
if(MSVC)
 target_compile_options(tu102_platform_model PRIVATE /W4)
else()
 target_compile_options(tu102_platform_model PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(platform_model_tests tests/test_platform_model.c)
target_link_libraries(platform_model_tests PRIVATE tu102_platform_model)
add_test(NAME platform_model_contracts COMMAND platform_model_tests)
add_executable(boot_tests tests/test_boot.c)
target_link_libraries(boot_tests PRIVATE tu102_host)
if(FUZZ)
 target_link_options(boot_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME boot_contracts COMMAND boot_tests)
